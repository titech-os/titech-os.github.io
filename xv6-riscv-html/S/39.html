 152 // Load an ELF program segment into pagetable at virtual address va.
 153 // va must be page-aligned
 154 // and the pages from va to va+sz must already be mapped.
 155 // Returns 0 on success, -1 on failure.
 156 static int
 157 loadseg(pagetable_t pagetable, uint64 va, struct inode *ip, uint offset,
 158         uint sz)
 159 {
 160   uint i, n;
 161   uint64 pa;
 162 
 163   for (i = 0; i < sz; i += PGSIZE) {
 164     pa = walkaddr(pagetable, va + i);
 165     if (pa == 0)
 166       panic("loadseg: address should exist");
 167     if (sz - i < PGSIZE)
 168       n = sz - i;
 169     else
 170       n = PGSIZE;
 171     if (readi(ip, 0, (uint64)pa, offset + i, n) != n)
 172       return -1;
 173   }
 174 
 175   return 0;
 176 }
