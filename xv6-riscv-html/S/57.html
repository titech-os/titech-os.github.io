  51 // this many virtio descriptors.
  52 // must be a power of two.
  53 #define NUM 8
  54 
  55 // a single descriptor, from the spec.
  56 struct virtq_desc {
  57   uint64 addr;
  58   uint32 len;
  59   uint16 flags;
  60   uint16 next;
  61 };
  62 #define VRING_DESC_F_NEXT  1 // chained with another descriptor
  63 #define VRING_DESC_F_WRITE 2 // device writes (vs read)
  64 
  65 // the (entire) avail ring, from the spec.
  66 struct virtq_avail {
  67   uint16 flags;     // always zero
  68   uint16 idx;       // driver will write ring[idx] next
  69   uint16 ring[NUM]; // descriptor numbers of chain heads
  70   uint16 unused;
  71 };
  72 
  73 // one entry in the "used" ring, with which the
  74 // device tells the driver about completed requests.
  75 struct virtq_used_elem {
  76   uint32 id; // index of start of completed descriptor chain
  77   uint32 len;
  78 };
  79 
  80 struct virtq_used {
  81   uint16 flags; // always zero
  82   uint16 idx;   // device increments when it adds a ring[] entry
  83   struct virtq_used_elem ring[NUM];
  84 };

  86 // these are specific to virtio block devices, e.g. disks,
  87 // described in Section 5.2 of the spec.
  88 
  89 #define VIRTIO_BLK_T_IN  0 // read the disk
  90 #define VIRTIO_BLK_T_OUT 1 // write the disk
  91 
  92 // the format of the first descriptor in a disk request.
  93 // to be followed by two more descriptors containing
  94 // the block, and a one-byte status.
  95 struct virtio_blk_req {
  96   uint32 type; // VIRTIO_BLK_T_IN or ..._OUT
  97   uint32 reserved;
  98   uint64 sector;
  99 };
