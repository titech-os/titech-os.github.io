 258 void
 259 iappend(uint inum, void *xp, int n)
 260 {
 261   char *p = (char *)xp;
 262   uint fbn, off, n1;
 263   struct dinode din;
 264   char buf[BSIZE];
 265   uint indirect[NINDIRECT];
 266   uint x;
 267 
 268   rinode(inum, &din);
 269   off = xint(din.size);
 270   // printf("append inum %d at off %d sz %d\n", inum, off, n);
 271   while (n > 0) {
 272     fbn = off / BSIZE;
 273     assert(fbn < MAXFILE);
 274     if (fbn < NDIRECT) {
 275       if (xint(din.addrs[fbn]) == 0) {
 276         din.addrs[fbn] = xint(freeblock++);
 277       }
 278       x = xint(din.addrs[fbn]);
 279     } else {
 280       if (xint(din.addrs[NDIRECT]) == 0) {
 281         din.addrs[NDIRECT] = xint(freeblock++);
 282       }
 283       rsect(xint(din.addrs[NDIRECT]), (char *)indirect);
 284       if (indirect[fbn - NDIRECT] == 0) {
 285         indirect[fbn - NDIRECT] = xint(freeblock++);
 286         wsect(xint(din.addrs[NDIRECT]), (char *)indirect);
 287       }
 288       x = xint(indirect[fbn - NDIRECT]);
 289     }
 290     n1 = min(n, (fbn + 1) * BSIZE - off);
 291     rsect(x, buf);
 292     bcopy(p, buf + off - (fbn * BSIZE), n1);
 293     wsect(x, buf);
 294     n -= n1;
 295     off += n1;
 296     p += n1;
 297   }
 298   din.size = xint(off);
 299   winode(inum, &din);
 300 }
